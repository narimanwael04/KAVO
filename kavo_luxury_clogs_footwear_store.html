Google Sheets & Gmail Backend Configuration

Direct automation for Kavo9320@gmail.com

How real orders & quotes reach your Google Sheets and Gmail:

  1. Create a new Google Sheet at sheets.new
  2. Click Extensions > Apps Script in your sheet.
  3. Paste the Google Apps Script code shown below.
  4. Click Deploy > New deployment > Select type: Web app.
  5. Set "Execute as: Me" and "Who has access: Anyone".
  6. Copy the Web App URL and paste it in the field below. Done!

If left empty, orders will gracefully open prefilled email to Kavo9320@gmail.com and WhatsApp confirmation.

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Type", "Reference", "Name", "Phone", "Gov", "Address", "Items Summary", "Total EGP", "Payment", "Notes"]);
    }
    
    var time = new Date().toLocaleString("en-US", {timeZone: "Africa/Cairo"});
    sheet.appendRow([
      time,
      data.kind || "Order",
      data.ref || ("KAVO-" + Date.now()),
      data.name || "",
      data.phone || "",
      data.gov || "",
      (data.city || "") + " - " + (data.addr || ""),
      data.summary || JSON.stringify(data.items || ""),
      data.total || 0,
      data.pay || "COD",
      data.notes || ""
    ]);

    // Send instant alert email to factory inbox:
    MailApp.sendEmail({
      to: "Kavo9320@gmail.com",
      subject: "New KAVO " + (data.kind || "Order") + " #" + (data.ref || ""),
      body: "New order received!\n\nCustomer: " + data.name + "\nPhone: " + data.phone + "\nGovernorate: " + data.gov + "\nTotal: " + data.total + " EGP\n\nItems:\n" + (data.summary || "") + "\n\nFull details recorded in your Google Sheet."
    });

    return ContentService.createTextOutput(JSON.stringify({status: "success", ref: data.ref}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
